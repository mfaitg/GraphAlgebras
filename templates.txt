relationsA := [ [pattern_1, replace_1], ..., [pattern_k, replace_k] ];


# In the functions below, replace the letter "A" EVERYWHERE by the actual name of your algebra.

applyRuleA := function(word)
   local i, pos, pattern, replace, prefix, suffix;
   for i in [ 1 .. Length(relationsA) ] do
      pattern := relationsA[i][1];
      replace := relationsA[i][2];
      pos := PositionWord(word, pattern, 1);
      if pos <> fail then
         prefix := Subword(word, 1, pos-1);
         suffix := Subword(word, pos+Length(pattern), Length(word));
         return (prefix^embA)*replace*(suffix^embA);
      fi;
   od;
   return word^embA;
end;

reduceA := function(P)
   local i, listP, wordsP, coeffsP, redP;
   listP := CoefficientsAndMagmaElements(P);
   wordsP := listP{ [ 1,3 .. Length(listP)-1 ] };
   coeffsP := listP{ [ 2,4 .. Length(listP) ] };
   Unbind(listP); #Delete local variable listP, not used in the sequel.
   
   redP := 0A;  #Zero element in the free algebra
   for i in [ 1 .. Length(wordsP) ] do
      redP := redP + coeffsP[i]*applyRuleA(wordsP[i]);
   od;
   if redP = P then
      return P;
   else
      return reduceA(redP);
   fi;
end;
